\chapter{Public-API Reference}
\label{app:api}

\section{Configuration}

\subsection{\texttt{config.py} and \texttt{wu/wu\_config.yaml}}
\begin{itemize}\setlength\itemsep{0pt}
\item Grid: \texttt{NX=134}, \texttt{NY=51}, \texttt{NW=9};
      \texttt{PLEVS} $=[1000,850,700,500,400,300,250,200,100]$\,hPa.
      Domain $85.5^\circ$N$\to$$10.5^\circ$N, $120^\circ$E$\to$$319.5^\circ$E.
\item \texttt{PIECES}: \texttt{surface} $\{1\}$, \texttt{lower} $\{2,3\}$,
      \texttt{upper} $\{4,5,6,7,8,9\}$ (mirrored authoritatively in
      \texttt{wu\_config.yaml} \texttt{pieces}).
\item Solver params live in \texttt{wu\_config.yaml} (\texttt{pass\_ab},
      \texttt{pass\_c}, \texttt{pass\_d}); \texttt{config.py} keeps grid,
      physical constants, the exact $10^{-8}$ Wu$\to$SI note, and paths.
\item Paths: \texttt{ERA5\_DIR}, \texttt{CLIM\_DIR}, \texttt{WU\_IN\_DIR},
      \texttt{WU\_OUT\_DIR}, \texttt{FIG\_DIR}.
\end{itemize}

\section{Fortran programs (\texttt{wu/fortran/})}
\begin{itemize}\setlength\itemsep{0pt}
\item \texttt{pvpialln\_94UV.f} --- Pass A/B: boundary $\theta$
      (\texttt{THB,THT}), interior Ertel PV, balanced $\psi,\Phi$
      (\Cref{sec:wu_passes}).  \texttt{PARAMETER (NW=9)}.
\item \texttt{qinvert21\_94.f} (BALNC) --- Pass C: total balanced
      inversion.  \texttt{PARAMETER (NL=9)} in main and \texttt{BALNC}.
\item \texttt{qinvertp21\_94.f} (BALP) --- Pass D: 3-piece perturbation
      inversion; \texttt{QLV} selects boundary $\theta$ (1 / \texttt{NL})
      and interior PV ($2..N_L{-}1$).  \texttt{PARAMETER (NL=9)} in main
      and \texttt{BALP}; hardcodes \texttt{MAX=8000, MAXT=2000,
      QMIN=1e-5, DPI=500/NL}.
\end{itemize}

\section{Shared front end (\texttt{shared\_steps/})}
\begin{itemize}\setlength\itemsep{0pt}
\item \texttt{01\_download/download\_era5.py} --- event hour at the 9
      levels (derived from \texttt{config.PLEVS}).
\item \texttt{03\_climatology/clim\_30yr\_mean.py} --- 30-yr mean +
      event subset to the 9 Wu levels (\texttt{mean\_clim.nc},
      \texttt{event.nc}).
\item \texttt{04\_write\_grid/write\_grid\_files.py} --- writes
      \texttt{mean.grid}, \texttt{event.grid} ($4N_W$ field-blocks).
\end{itemize}

\section{Wu step drivers (\texttt{wu/steps/})}
\begin{itemize}\setlength\itemsep{0pt}
\item \texttt{05\_wu\_pass\_ab/} --- compile + Pass A/B; writes
      \texttt{meanq}, \texttt{meanh}, \texttt{event\_\{q,h\}.out}.
\item \texttt{06\_wu\_pass\_c/} --- Pass C; writes \texttt{event\_bal.out}.
\item \texttt{07\_wu\_pass\_d/} --- Pass D; writes \texttt{event\_pert.out}
      (3 pieces, \texttt{QLV} from \texttt{wu\_config.yaml}).
\item \texttt{08\_parse\_outputs/parse\_and\_pv.py} --- Wu $Q\to$SI
      ($\times10^{-8}$); writes \texttt{piecewise\_psi.nc},
      \texttt{pv\_advection.nc} (all SI).
\item \texttt{10\_fig8/fig8\_replica.py} --- Davis-style PV-advection
      diagnostic (surface/lower/upper).
\item \texttt{11\_xsection/} --- \texttt{xsection\_vinduced.py} (zonal
      cross-section), \texttt{xsection\_3d\_perpendicular.py} (3-D
      vertical structure, Fig.-16 style), \texttt{upper\_psi\_by\_piece.py},
      \texttt{closure\_check.py}, \texttt{closure\_diagnostic.py}.
\end{itemize}

\section{Key output: \texttt{piecewise\_psi.nc}}
\begin{itemize}\setlength\itemsep{0pt}
\item Dims \texttt{(piece, plev, lat, lon)} $=(3, 9, 51, 134)$; the
      \texttt{piece} axis is ordered \texttt{surface, lower, upper}.
\item Variables \texttt{psi}, \texttt{geopotential}, \texttt{U\_induced},
      \texttt{V\_induced}, all SI.
\end{itemize}
